-- Actualización de cabecera y detalle de una cotización

DECLARE @TOTAL DECIMAL(18,2),
        @SUBTOTAL DECIMAL(18,2),
        @IMPUESTO DECIMAL(18,2),
        @COD_COTIZACION CHAR(15)

SET @COD_COTIZACION = 'COT-000123'

SET @TOTAL = (
    SELECT SUM(DET_IMPORTE)
    FROM dbo.DetalleCotizacion
    WHERE COTIZACION_ID = @COD_COTIZACION
)

SET @SUBTOTAL = @TOTAL / 1.18
SET @IMPUESTO = @TOTAL - @SUBTOTAL

UPDATE dbo.Cotizacion
SET
    COT_SUBTOTAL = @SUBTOTAL,
    COT_IMPUESTO = @IMPUESTO,
    COT_TOTAL = @TOTAL
WHERE COTIZACION_ID = @COD_COTIZACION

SELECT *
FROM dbo.Cotizacion
WHERE COTIZACION_ID = @COD_COTIZACION

-- Vista para consolidar información de cuentas por cobrar

CREATE VIEW [dbo].[V_CUENTAS_POR_COBRAR]
AS

SELECT
    a.EmpresaId,
    a.SucursalId,
    a.ClienteId,
    a.AuxiliarId,
    c.NombreAuxiliar,
    a.TipoDocumento,
    a.NumeroDocumento,
    a.FechaEmision,

    a.FechaVencimiento
        + ISNULL(f.DiasGracia, 0)
        AS FechaVencimiento,

    DATEDIFF(
        DAY,
        a.FechaEmision,
        a.FechaVencimiento
        + ISNULL(f.DiasGracia, 0)
    ) AS DiasPlazo,

    DATEDIFF(
        DAY,
        a.FechaVencimiento,
        GETDATE()
    ) AS DiasVencidos,

    a.TipoMoneda,
    a.ImporteDocumento,
    a.ImporteAbonado,

    ABS(a.ImporteCuenta)
        - ABS(a.ImporteAbonado)
        AS SaldoDocumento,

    a.SaldoCuenta,
    a.SaldoLetras,
    a.SaldoResidual,

    CASE a.TipoMoneda

        WHEN 'PEN'
            THEN a.SaldoCuenta

        ELSE
            dbo.fn_ConvertirMoneda(
                a.SaldoCuenta,
                GETDATE(),
                'PEN'
            )

    END AS SaldoMonedaLocal,

    ISNULL(
        v.NombreVendedor,
        ''
    ) AS NombreVendedor

FROM dbo.CuentaCorriente a

INNER JOIN dbo.CuentasCobrar b
    ON b.EmpresaId = a.EmpresaId
    AND b.DocumentoId = a.DocumentoId

INNER JOIN dbo.Auxiliares c
    ON c.EmpresaId = a.EmpresaId
    AND c.AuxiliarId = a.AuxiliarId

LEFT JOIN dbo.Clientes f
    ON f.EmpresaId = a.EmpresaId
    AND f.AuxiliarId = a.AuxiliarId

LEFT JOIN dbo.Vendedores v
    ON v.EmpresaId = b.EmpresaId
    AND v.VendedorId = b.VendedorId

WHERE
    a.IndicadorEstado = '1'
    AND (
        ISNULL(a.SaldoCuenta, 0) <> 0
        OR
        ISNULL(a.SaldoLetras, 0) <> 0
    );

-- Creación de usuario con control transaccional

CREATE PROCEDURE sp_CrearUsuario
(
    @nombre VARCHAR(150),
    @apellido VARCHAR(150),
    @documento CHAR(15),
    @correo VARCHAR(250)
)
AS
BEGIN

    BEGIN TRY

        BEGIN TRANSACTION

        INSERT INTO dbo.Personas
        (
            Documento,
            Nombre,
            Apellido,
            Estado
        )
        VALUES
        (
            @documento,
            @nombre,
            @apellido,
            'ACTIVO'
        )

        INSERT INTO dbo.Usuarios
        (
            Correo,
            CodigoEmpresa,
            PasswordHash,
            Estado,
            RolId,
            FechaCreacion,
            Documento
        )
        VALUES
        (
            @correo,
            'EMPRESA01',
            '[HASH_PROTEGIDO]',
            1,
            3,
            GETDATE(),
            @documento
        )

        COMMIT TRANSACTION

    END TRY

    BEGIN CATCH

        ROLLBACK TRANSACTION
        THROW

    END CATCH

END


-- Función para calcular stock disponible

CREATE FUNCTION dbo.fn_StockDisponible
(
    @empresa CHAR(2),
    @almacen CHAR(4),
    @producto CHAR(13)
)

RETURNS NUMERIC(16,3)

AS
BEGIN

    DECLARE @stockDisponible NUMERIC(16,3)
    DECLARE @stockFisico NUMERIC(16,3)
    DECLARE @stockReservado NUMERIC(16,3)

    SET @stockFisico =
        dbo.fn_StockFisicoAlmacen(
            @empresa,
            @almacen,
            @producto
        )

    SET @stockReservado =
        dbo.fn_StockReservado(
            @empresa,
            @almacen,
            @producto
        )

    SET @stockDisponible =
        ISNULL(@stockFisico, 0)
        -
        ISNULL(@stockReservado, 0)

    RETURN @stockDisponible

END

-- Vista para consolidar información de inventario
-- Permite consultar productos con lote, serie o referencia

CREATE VIEW dbo.V_ActivosInventario
AS

SELECT DISTINCT

    A.EmpresaId AS Empresa,
    A.ProductoId AS CodigoProducto,
    I.NombreProducto,
    L.DescripcionTipoInventario AS TipoInventario,
    J.DescripcionMarca AS Marca,
    I.Modelo,
    K.DescripcionFamilia AS Familia,

    CASE
        WHEN A.TipoRegistro = '1'
            THEN 'LOTE'

        WHEN A.TipoRegistro = '2'
            THEN 'SERIE'

        WHEN A.TipoRegistro = '3'
            THEN 'REFERENCIA'

        ELSE
            'SIN INDICADOR'
    END AS Indicador,

    CASE
        WHEN A.TipoRegistro = '1'
            THEN D.NumeroLote

        WHEN A.TipoRegistro = '2'
            THEN B.NumeroSerie

        WHEN A.TipoRegistro = '3'
            THEN F.NumeroReferencia

        ELSE
            'SIN INDICADOR'
    END AS Identificador,

    CASE
        WHEN A.TipoRegistro = '1'
            THEN D.SaldoLote

        WHEN A.TipoRegistro = '2'
            THEN B.SaldoSerie

        WHEN A.TipoRegistro = '3'
            THEN F.SaldoReferencia

        ELSE
            A.SaldoProducto
    END AS Saldo

FROM dbo.SaldosProductos A

LEFT JOIN dbo.SaldosSeries B
    ON A.EmpresaId = B.EmpresaId
    AND A.ProductoId = B.ProductoId

LEFT JOIN dbo.ProductosSeries C
    ON B.EmpresaId = C.EmpresaId
    AND B.ProductoId = C.ProductoId

LEFT JOIN dbo.SaldosLotes D
    ON A.EmpresaId = D.EmpresaId
    AND A.ProductoId = D.ProductoId

LEFT JOIN dbo.LotesProductos E
    ON D.ProductoId = E.ProductoId

LEFT JOIN dbo.SaldosReferencias F
    ON A.EmpresaId = F.EmpresaId
    AND A.ProductoId = F.ProductoId

LEFT JOIN dbo.Almacenes H
    ON A.EmpresaId = H.EmpresaId

LEFT JOIN dbo.Productos I
    ON A.EmpresaId = I.EmpresaId
    AND A.ProductoId = I.ProductoId

LEFT JOIN dbo.Marcas J
    ON I.EmpresaId = J.EmpresaId

LEFT JOIN dbo.FamiliasProductos K
    ON I.EmpresaId = K.EmpresaId

LEFT JOIN dbo.TiposInventario L
    ON I.TipoInventarioId = L.TipoInventarioId

WHERE
    A.EmpresaId = '01'
    AND I.TipoInventarioId = '10';

    IdProducto,
    Codigo,
    Descripcion,
    Stock

FROM OPENQUERY(
    MYSQLSERVER,

    '
        SELECT
            IdProducto,
            Codigo,
            Descripcion,
            Stock

        FROM productos

        WHERE estado = 1
    '
