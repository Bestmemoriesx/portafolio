-- Actualización de cabecera y detalle de una cotización

-- //////// ACTUALIZAR CABECERA ////////////

DECLARE @TOTAL DECIMAL(18,2),
        @SUBTOTAL DECIMAL(18,2),
        @IMPUESTO DECIMAL(18,2),
        @COD_COTIZACION CHAR(15)

SET @COD_COTIZACION = 'COT-000123'

SET @TOTAL = (
    SELECT SUM(DET_IMPORTE)
    FROM dbo.DetalleCotizacion
    WHERE COTIZACION_ID = @COD_COTIZACION
)

SET @SUBTOTAL = @TOTAL / 1.18
SET @IMPUESTO = @TOTAL - @SUBTOTAL

UPDATE dbo.Cotizacion
SET
    COT_SUBTOTAL = @SUBTOTAL,
    COT_IMPUESTO = @IMPUESTO,
    COT_TOTAL = @TOTAL
WHERE COTIZACION_ID = @COD_COTIZACION

SELECT *
FROM dbo.Cotizacion
WHERE COTIZACION_ID = @COD_COTIZACION

-- //////// ACTUALIZAR DETALLE ////////////

DECLARE @PRECIO DECIMAL(18,2),
        @TOTAL_DETALLE DECIMAL(18,2),
        @PRECIO_CONVERTIDO DECIMAL(18,2),
        @DESCUENTO DECIMAL(18,2),
        @COD_PRODUCTO CHAR(10),
        @CANTIDAD INT,
        @TIPO_CAMBIO DECIMAL(18,2)

SET @TIPO_CAMBIO = (
    SELECT TOP 1 TIPO_CAMBIO
    FROM dbo.TiposCambio
    WHERE CONVERT(DATE, FECHA_CAMBIO)
        = CONVERT(DATE, GETDATE())
)

SET @COD_PRODUCTO = 'PROD-001'
SET @COD_COTIZACION = 'COT-000123'
SET @PRECIO = 11500.00

SET @CANTIDAD = (
    SELECT TOP 1 CANTIDAD
    FROM dbo.DetalleCotizacion
    WHERE PRODUCTO_ID = @COD_PRODUCTO
      AND COTIZACION_ID = @COD_COTIZACION
)

SET @TOTAL_DETALLE = @PRECIO * @CANTIDAD
SET @PRECIO_CONVERTIDO = @TOTAL_DETALLE * @TIPO_CAMBIO
SET @DESCUENTO = 30

UPDATE dbo.DetalleCotizacion
SET
    PRECIO_UNITARIO = @PRECIO,
    IMPORTE_TOTAL = @TOTAL_DETALLE,
    IMPORTE_CONVERTIDO = @PRECIO_CONVERTIDO,
    DESCUENTO = @DESCUENTO,
    CANTIDAD = @CANTIDAD
WHERE COTIZACION_ID = @COD_COTIZACION
  AND PRODUCTO_ID = @COD_PRODUCTO

SELECT *
FROM dbo.DetalleCotizacion
WHERE COTIZACION_ID = @COD_COTIZACION

-- Vista para consolidar información de cuentas por cobrar

CREATE VIEW [dbo].[V_CUENTAS_POR_COBRAR]
AS

SELECT
    a.EmpresaId,
    a.SucursalId,
    a.ClienteId,
    a.AuxiliarId,
    c.NombreAuxiliar,
    a.TipoDocumento,
    a.NumeroDocumento,
    a.FechaEmision,

    a.FechaVencimiento
        + ISNULL(f.DiasGracia, 0) AS FechaVencimiento,

    e.DescripcionCondicion,

    DATEDIFF(
        DAY,
        a.FechaEmision,
        a.FechaVencimiento + ISNULL(f.DiasGracia, 0)
    ) AS DiasPlazo,

    DATEDIFF(
        DAY,
        a.FechaVencimiento + ISNULL(f.DiasGracia, 0),
        GETDATE()
    ) AS DiasVencidos,

    a.TipoMoneda,
    a.ImporteDocumento,
    a.ImporteAbonado,

    ABS(a.ImporteCuenta)
        - ABS(a.ImporteAbonado) AS SaldoDocumento,

    a.SaldoCuenta,
    a.SaldoLetras,
    a.SaldoResidual,

    a.SaldoCuenta
        + a.SaldoLetras
        + a.SaldoResidual AS SaldoReal,

    a.ImporteRetencion,
    a.ImporteIGV,
    a.GlosaDocumento,

    CASE a.TipoMoneda
        WHEN 'PEN'
            THEN a.SaldoCuenta
        ELSE
            dbo.fn_ConvertirMoneda(
                a.SaldoCuenta,
                GETDATE(),
                'PEN'
            )
    END AS SaldoMonedaLocal,

    CASE a.TipoMoneda
        WHEN 'USD'
            THEN a.SaldoCuenta
        ELSE
            dbo.fn_ConvertirMoneda(
                a.SaldoCuenta,
                GETDATE(),
                'USD'
            )
    END AS SaldoDolares,

    CASE
        WHEN b.EstadoDocumento = '5'
             AND d.ClaseCuenta = '2'
             AND a.GlosaDocumento LIKE '%descuento%'
        THEN '6'
        ELSE b.EstadoDocumento
    END AS EstadoDocumento,

    CASE
        WHEN a.TipoDocumento = 'LET'
        THEN
            CASE
                WHEN b.EstadoDocumento IN ('0', '5')
                THEN
                    RTRIM(
                        ISNULL(k.DescripcionEstado, '')
                    )
                ELSE
                    RTRIM(
                        ISNULL(k.DescripcionEstado, '')
                    )
                    + ' - '
                    + RTRIM(
                        ISNULL(j.DescripcionLinea, '')
                    )
            END
        ELSE ''
    END AS DescripcionSituacion,

    ISNULL(v.NombreVendedor, '') AS NombreVendedor,
    ISNULL(n.NombreBanco, '') AS NombreBanco,

    b.FechaDescuento,

    CASE
        WHEN b.FechaDescuento IS NOT NULL
             AND b.FechaDescuento > a.FechaEmision
        THEN b.FechaDescuento
        ELSE a.FechaEmision
    END AS FechaInicio,

    b.ImporteComision,

    CONVERT(
        NUMERIC(10,2),
        ROUND(
            CASE
                WHEN ISNULL(a.ImporteDocumento, 0) <> 0
                     AND ISNULL(b.ImporteComision, 0) > 0
                THEN
                    (
                        ABS(a.SaldoCuenta)
                        * ISNULL(b.ImporteComision, 0)
                    ) / a.ImporteDocumento
                ELSE 0
            END,
            2
        )
    ) AS ComisionSaldo

FROM dbo.CuentaCorriente a WITH (NOLOCK)

INNER JOIN dbo.CuentasCobrar b WITH (NOLOCK)
    ON b.EmpresaId = a.EmpresaId
    AND b.SucursalId = a.SucursalId
    AND b.DocumentoId = a.DocumentoId

INNER JOIN dbo.Auxiliares c
    ON c.EmpresaId = a.EmpresaId
    AND c.AuxiliarId = a.AuxiliarId

INNER JOIN dbo.CuentasContables d WITH (NOLOCK)
    ON d.EmpresaId = a.EmpresaId
    AND d.CuentaId = a.CuentaId

LEFT JOIN dbo.CondicionesPago e WITH (NOLOCK)
    ON e.CondicionId = a.CondicionPagoId

LEFT JOIN dbo.Clientes f WITH (NOLOCK)
    ON f.EmpresaId = a.EmpresaId
    AND f.AuxiliarId = a.AuxiliarId

LEFT JOIN dbo.TiposMoneda h WITH (NOLOCK)
    ON h.TipoMoneda = a.TipoMoneda

LEFT JOIN dbo.LineasCredito j WITH (NOLOCK)
    ON j.EmpresaId = b.EmpresaId
    AND j.LineaCredito = b.LineaCredito

LEFT JOIN dbo.EstadosDocumento k WITH (NOLOCK)
    ON k.EstadoDocumento = b.EstadoDocumento

LEFT JOIN dbo.Vendedores v WITH (NOLOCK)
    ON v.EmpresaId = b.EmpresaId
    AND v.VendedorId = b.VendedorId

LEFT JOIN dbo.BancosCuentas m WITH (NOLOCK)
    ON m.EmpresaId = b.EmpresaId
    AND m.CuentaBancaria = b.CuentaBancaria

LEFT JOIN dbo.Bancos n WITH (NOLOCK)
    ON n.BancoId = m.BancoId

WHERE
    a.IndicadorEstado = '1'
    AND (
        ISNULL(a.SaldoCuenta, 0) <> 0
        OR ISNULL(a.SaldoLetras, 0) <> 0
        OR ISNULL(a.SaldoResidual, 0) <> 0
    )
    AND a.TipoPago = 'C';

-- Consulta reutilizable para solicitudes recurrentes

CREATE OR ALTER PROCEDURE
dbo.usp_ConsultaVentas

    @FechaInicio DATE,
    @FechaFin DATE

AS

BEGIN

    DECLARE
        @Estado VARCHAR(20);

    SET
        @Estado = 'ACTIVO';

    SELECT
        IdVenta,
        Fecha,
        Total

    FROM dbo.Ventas

    WHERE
        Fecha BETWEEN
            @FechaInicio
            AND
            @FechaFin

        AND
            Estado = @Estado;

END;

-- Obtener el registro más reciente por cliente

WITH UltimasVentas AS
(

    SELECT
        IdCliente,
        IdVenta,
        Fecha,
        Total,

        ROW_NUMBER() OVER
        (
            PARTITION BY IdCliente
            ORDER BY Fecha DESC
        ) AS Orden

    FROM dbo.Ventas

)

SELECT
    IdCliente,
    IdVenta,
    Fecha,
    Total

FROM UltimasVentas

WHERE
    Orden = 1;
