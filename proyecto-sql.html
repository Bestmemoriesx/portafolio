-- Integración de información proveniente de distintas tablas

SELECT
    v.IdVenta,
    v.Fecha,
    c.NombreCliente,
    v.Total,

    ISNULL(
        c.TipoCliente,
        'NO DEFINIDO'
    ) AS TipoCliente,

    CASE
        WHEN v.Total > 10000
            THEN 'ALTA'
        ELSE 'NORMAL'
    END AS NivelVenta

FROM Ventas.dbo.Ventas v

LEFT JOIN Clientes.dbo.Clientes c
    ON c.IdCliente = v.IdCliente

LEFT JOIN Productos.dbo.Productos p
    ON p.IdProducto = v.IdProducto

WHERE v.Estado = 'ACTIVO';

-- Vista preparada para reutilizar información

CREATE VIEW dbo.vw_VentasIntegradas

AS

SELECT
    v.IdVenta,
    v.Fecha,
    c.NombreCliente,
    p.Descripcion AS Producto,
    v.Cantidad,
    v.Total

FROM Ventas.dbo.Ventas v

INNER JOIN Clientes.dbo.Clientes c
    ON c.IdCliente = v.IdCliente

INNER JOIN Productos.dbo.Productos p
    ON p.IdProducto = v.IdProducto

WHERE v.Anulado = 0;

-- Consulta reutilizable para solicitudes recurrentes

CREATE OR ALTER PROCEDURE
dbo.usp_ConsultaVentas

    @FechaInicio DATE,
    @FechaFin DATE

AS

BEGIN

    DECLARE
        @Estado VARCHAR(20);

    SET
        @Estado = 'ACTIVO';

    SELECT
        IdVenta,
        Fecha,
        Total

    FROM dbo.Ventas

    WHERE
        Fecha BETWEEN
            @FechaInicio
            AND
            @FechaFin

        AND
            Estado = @Estado;

END;

-- Obtener el registro más reciente por cliente

WITH UltimasVentas AS
(

    SELECT
        IdCliente,
        IdVenta,
        Fecha,
        Total,

        ROW_NUMBER() OVER
        (
            PARTITION BY IdCliente
            ORDER BY Fecha DESC
        ) AS Orden

    FROM dbo.Ventas

)

SELECT
    IdCliente,
    IdVenta,
    Fecha,
    Total

FROM UltimasVentas

WHERE
    Orden = 1;

FROM OPENQUERY(
    MYSQLSERVER,

            IdProducto,

        FROM productos
